cmake_minimum_required(VERSION 3.20)
project(MatchEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(orderbook orderbook.cpp pipeline.cpp itch.cpp)
target_link_libraries(orderbook PUBLIC Threads::Threads)
target_include_directories(orderbook PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
option(ITCH_DISABLE_AVX2 "Force portable scalar decoding" OFF)
if(ITCH_DISABLE_AVX2)
  target_compile_definitions(orderbook PRIVATE ITCH_DISABLE_AVX2)
endif()
if(MSVC)
  target_compile_options(orderbook PRIVATE /W4)
else()
  target_compile_options(orderbook PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(match-engine main.cpp)
target_link_libraries(match-engine PRIVATE orderbook)
include(CTest)
if(BUILD_TESTING)
  add_executable(engine-tests tests.cpp)
  target_link_libraries(engine-tests PRIVATE orderbook)
  add_test(NAME engine-tests COMMAND engine-tests)
  set_tests_properties(engine-tests PROPERTIES TIMEOUT 90)
endif()
add_executable(stress-benchmark stress_benchmark.cpp)
target_link_libraries(stress-benchmark PRIVATE orderbook)
option(WITH_GOOGLE_BENCHMARK "Build optional Google Benchmark suite" OFF)
if(WITH_GOOGLE_BENCHMARK)
  find_package(benchmark CONFIG REQUIRED)
  add_executable(benchmark-orderbook benchmark_orderbook.cpp)
  target_link_libraries(benchmark-orderbook PRIVATE orderbook benchmark::benchmark)
endif()
